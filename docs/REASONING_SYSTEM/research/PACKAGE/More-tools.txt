import numpy as np
def generate_harmonic_data(N=200, noise=0.01, seed=0):
    np.random.seed(seed)
    A = 1.0
    omega = 2.0
    t = np.linspace(0, 4*np.pi, N)
    x = A * np.cos(omega * t)
    x_noisy = x + noise * np.random.randn(N)
    train_mask = t <= 2*np.pi
    test_mask = t > 2*np.pi
    return {
        "t_train": t[train_mask],
        "y_train": x_noisy[train_mask],
        "t_test": t[test_mask],
        "y_test": x[test_mask],
        "t_full": t,
        "y_true": x
    }

🔹 Add model wrapper (reuse structure)
class HarmonicHybridModel:
    def __init__(self):
        self.A = None
        self.omega = None
    def fit(self, t, y):
        import numpy as np
        # crude frequency estimation
        fft = np.fft.fft(y)
        freqs = np.fft.fftfreq(len(y), d=(t[1]-t[0]))
        idx = np.argmax(np.abs(fft[1:])) + 1
        self.omega = 2*np.pi*abs(freqs[idx])
        self.A = np.max(y)
    def predict(self, t):
        import numpy as np
        return self.A * np.cos(self.omega * t)
